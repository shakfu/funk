-- funk-midi --audio --voices 2 --no-sleep --cycle 2000 --script (tests/CMakeLists.txt):
-- control patterns (lib/control.fk), as SuperDirt's MIDI target plays them,
-- on MIDI and on the voices. The sum of the gates is printed: velocity 50
-- of 127 a note, from amp's 0.4. pat is redefined 100 ms before a cycle.
sig = voices (\gate freq -> gate)
-- note 0 is MIDI 60; a CC before each note, at its time
pat = note "0 [4 7]" # ccn "74" # ccv "<64 100>"
:wait 1900
-- channel 2 (midichan 1), velocity from gain, a program change and a bend;
-- |+ adds 2 to n in the second cycle; legato halves the notes
pat = n "0 12" # midichan "1" # gain "1.2" # progNum "3" # midibend "8192" # legato "0.5" |+ n "<0 2>"
:wait 4000
-- a sample's event sends no MIDI; a Pattern Int plays as before
pat = s "bd sn"
:wait 2000
pat = p.mini "72"
:wait 1000
:wait 1000
-- a voice reads its event's parameters: its gate times level, which the
-- event sets before the gate rises; the graph has no cutoff, which is skipped
sig = voicesWith (\gate freq param -> gate * param "level" 1.0)
pat = note "0 4" # pF "level" (p.miniF "<0.5 0.25>") # cutoff "300"
:wait 4000
-- samples by name, as d.kit's controls: bd:1 is b.wav, sn x.WAV; level is
-- (min gain 2)^4, so gain 0.5 plays at 1/16; nosuch has no controls
sig = d.kit "tests/hosts/samples" ["bd", "bd:1", "sn"]
pat = s "bd bd:1 sn nosuch" # gain "<1 0.5>"
:wait 4000
-- a name's events overlap on its voices: bd's last 0.05 s from its middle,
-- every 31.25 ms, sum to about 0.5. With a cut group each stops the one
-- before in a 1 ms linear fade as the next fades in on a sine, as SuperDirt's,
-- and the two sum to at most 1.105 times one: 0.276
pat = s "bd*64" # begin "0.5"
:wait 2000
pat = s "bd*64" # begin "0.5" # cut "1"
:wait 2000
